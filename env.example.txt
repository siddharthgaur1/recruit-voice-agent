# --- LLM provider -------------------------------------------------------
# "groq" (default), "gemini", or "ollama" (local, no key needed).
LLM_PROVIDER=groq

# Free tier key: https://console.groq.com/keys
GROQ_API_KEY=
# Escalation tier (higher accuracy, higher latency). Re-check
# https://console.groq.com/docs/models if this model is ever decommissioned
# (llama-3.3-70b-versatile, the original pick, already was).
GROQ_MODEL=openai/gpt-oss-120b
# Fast tier (tried first; escalates to GROQ_MODEL only on low confidence).
GROQ_MODEL_FAST=openai/gpt-oss-20b
# gpt-oss models only: "low"/"medium"/"high". Ignored for non-gpt-oss models.
GROQ_REASONING_EFFORT=low
# Set false to always call GROQ_MODEL directly, skipping the fast tier.
EXTRACTION_ESCALATION_ENABLED=true

# Only used when LLM_PROVIDER=gemini. Free tier key:
# https://aistudio.google.com/apikey
GEMINI_API_KEY=
GEMINI_MODEL=gemini-3.6-flash

# Only used when LLM_PROVIDER=ollama. No key needed -- install Ollama
# (https://ollama.com) and `ollama pull llama3.2` first. See docs/RESULTS.md
# for why this isn't the default (CPU-only inference was measured too slow).
OLLAMA_MODEL=llama3.2:latest
OLLAMA_HOST=http://localhost:11434

# --- Agent behavior ------------------------------------------------------
COMPANY_NAME=Acme Corp
DISCLOSURE_TEXT=Hi, this is an automated recruiting assistant calling on behalf of {company}.

RECORDING_ENABLED=false
DB_PATH=recruit_agent.db

# --- Phase 3: local voice loop (faster-whisper + Piper, both free/local) -
# Multilingual (not e.g. "small.en") -- needed for Hinglish code-switching.
WHISPER_MODEL=small
WHISPER_DEVICE=cpu
WHISPER_COMPUTE_TYPE=int8

# Download with: python -m piper.download_voices en_US-lessac-medium --download-dir models
PIPER_MODEL_PATH=models/en_US-lessac-medium.onnx

# 0-3, higher = more aggressive about filtering non-speech.
VAD_AGGRESSIVENESS=2

# Trailing silence (ms) before an utterance is considered finished.

# Seconds of silence before "Are you still there?" / before giving up.
SILENCE_PROMPT_AFTER_S=7.0
SILENCE_CLOSE_AFTER_S=12.0
